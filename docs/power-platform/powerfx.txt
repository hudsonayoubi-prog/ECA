// ===== 6.5 Leaderboard calculation =====
Set(varRules, LookUp('Scoring Rule Sets', 'Is Current' = true));
ClearCollect(colCounted, Filter('Audit Visits',
    'Reporting Month' = varMonth, 'Visit Status' = 'Visit Status'.Completed,
    'Approval Status' = 'Approval Status'.Approved, Imported = false));

ClearCollect(colRaw, ForAll(
  Filter('ECA Users', Role = 'ECA Role'.'Student Marketeer', Status = 'Active / Inactive'.Active) As u,
  With({mine: Filter(colCounted, 'Staff Member'.'ECA User' = u.'ECA User')}, {
    User: u, Name: u.'Full Name', StateCode: u.State.'State Code', Manager: u.'Reports To'.'Full Name',
    Audits: CountRows(mine),
    // ranging: latest approved audit per location, averaged
    Ranging: Coalesce(Average(ForAll(Distinct(mine, Location.Location) As l,
        First(Sort(Filter(mine, Location.Location = l.Value), 'Visit Date', SortOrder.Descending)).'Ranging %'), Value), 0),
    TDP: Sum(mine, 'DPs Added'), Editions: Sum(mine, 'Editions Sold In'),
    POS: Sum(mine, 'POS Placed'), Coolers: Sum(mine, 'Coolers Installed'),
    LastAt: Max(mine, 'Approved On')
  })));

Set(varMin, varRules.'Minimum Approved Audits');
Set(varMax, With({q: Filter(colRaw, Audits >= varMin)},
    {TDP: Max(q, TDP), Ed: Max(q, Editions), POS: Max(q, POS), Cool: Max(q, Coolers)}));
Set(varTarget, varRules.Normalisation = 'Normalisation'.'Against targets');

ClearCollect(colScored, ForAll(colRaw As r, With({
    sT: If(varTarget, Min(100, 100 * r.TDP / Max(1, varRules.'Target TDP')), If(varMax.TDP > 0, 100 * r.TDP / varMax.TDP, 0)),
    sE: If(varTarget, Min(100, 100 * r.Editions / Max(1, varRules.'Target Editions')), If(varMax.Ed > 0, 100 * r.Editions / varMax.Ed, 0)),
    sP: If(varTarget, Min(100, 100 * r.POS / Max(1, varRules.'Target POS')), If(varMax.POS > 0, 100 * r.POS / varMax.POS, 0)),
    sC: If(varTarget, Min(100, 100 * r.Coolers / Max(1, varRules.'Target Coolers')), If(varMax.Cool > 0, 100 * r.Coolers / varMax.Cool, 0))},
  Patch(r, {Qualified: r.Audits >= varMin, sRanging: r.Ranging, sTDP: sT, sEd: sE, sPOS: sP, sCool: sC,
    Total: If(r.Audits >= varMin, Round((r.Ranging * varRules.'Weight Ranging' + sT * varRules.'Weight TDP' + sE * varRules.'Weight Editions'
        + sP * varRules.'Weight POS' + sC * varRules.'Weight Coolers') / 100, 1), 0)}))));

With({s: SortByColumns(colScored, Qualified, SortOrder.Descending, Total, SortOrder.Descending,
        TDP, SortOrder.Descending, Ranging, SortOrder.Descending, Audits, SortOrder.Descending, LastAt, SortOrder.Ascending, Name, SortOrder.Ascending)},
  ClearCollect(colBoard, ForAll(Sequence(CountRows(s)) As i,
    Patch(Index(s, i.Value), {Rank: If(Index(s, i.Value).Qualified, i.Value, Blank())}))));

// ===== 6.6 Finalise and send results (Results tab, Leads and Admins) =====
Set(varSnap, Patch('Leaderboard Snapshots',
    Coalesce(LookUp('Leaderboard Snapshots', Month = varMonth), Defaults('Leaderboard Snapshots')),
    {Month: varMonth, Status: 'Snapshot Status'.Final, 'Finalised By': gblMe, 'Finalised On': Now(), 'Rule Set': varRules,
     'Unapproved Audits Excluded': CountRows(Filter('Audit Visits', 'Reporting Month' = varMonth, 'Approval Status' = 'Approval Status'.Pending))}));
ForAll(colBoard As r, Patch('Snapshot Rows', Defaults('Snapshot Rows'), {
    Name: varMonth & " – " & r.Name, Snapshot: varSnap, 'Student Marketeer': r.User, Rank: r.Rank, 'Total Score': r.Total,
    Qualified: r.Qualified, 'Approved Audits': r.Audits, 'Ranging %': r.Ranging, 'TDP Added': r.TDP, 'Editions Sold': r.Editions,
    'POS Placed': r.POS, 'Coolers Added': r.Coolers, 'Score Ranging': r.sRanging, 'Score TDP': r.sTDP,
    'Score Editions': r.sEd, 'Score POS': r.sPOS, 'Score Coolers': r.sCool}));
Patch('ECA Audit Logs', Defaults('ECA Audit Logs'), {Action: "Finalised leaderboard", Category: 'Log Category'.Leaderboard,
    User: gblMe, 'Logged On': Now(), Record: varMonth, 'New Value': "Final"});

// ===== 6.6 Finalise and send results (Results tab, Leads and Admins) =====
Patch('Results Distributions', Defaults('Results Distributions'), {
    Name: varMonth, Snapshot: varSnap, 'Generated On': Now(), 'Distributed By': gblMe,
    Method: drpMethod.Selected.Value, Message: txtMessage.Text})
